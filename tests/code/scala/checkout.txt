package shop

import scala.annotation.tailrec
import scala.concurrent.{ExecutionContext, Future}

/** An order, priced in cents. */
case class Order(id: Long, items: List[String], cents: Int)

sealed trait Payment
object Payment {
  case object Cash extends Payment
  final case class Card(number: String) extends Payment
}

trait Pricing {
  def total(order: Order): BigDecimal = BigDecimal(order.cents) / 100
}

object Checkout extends Pricing {
  private val MaxItems = 50
  val Banner = """
    |Thanks for "shopping"
    |""".stripMargin

  def describe(order: Order)(implicit ec: ExecutionContext): Future[String] = Future {
    val count = order.items.size
    s"Order ${order.id} has $count items, total ${total(order)}"
  }

  def pay(payment: Payment): String = payment match {
    case Payment.Cash         => "cash"
    case Payment.Card(number) => f"card ending ${number.takeRight(4)}%s"
  }

  @tailrec
  def gcd(a: Int, b: Int): Int = if (b == 0) a else gcd(b, a % b)

  val doubled = List(1, 2, 3).map(_ * 2).filter(_ > 0x2)
  val ch = 'c'
  val `type` = null
}
