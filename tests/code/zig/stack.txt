const std = @import("std");
const Allocator = std.mem.Allocator;

/// A stack of items, grown as needed.
pub fn Stack(comptime T: type) type {
    return struct {
        items: std.ArrayList(T),

        const Self = @This();

        pub fn init(allocator: Allocator) Self {
            return .{ .items = std.ArrayList(T).init(allocator) };
        }

        pub fn push(self: *Self, item: T) !void {
            try self.items.append(item);
        }

        pub fn pop(self: *Self) ?T {
            return self.items.popOrNull();
        }
    };
}

const banner =
    \\Zig's multiline strings
    \\keep "quotes" as written
;

test "stack pushes and pops" {
    var stack = Stack(u32).init(std.testing.allocator);
    defer stack.items.deinit();
    try stack.push(0xFF);
    const top = stack.pop() orelse unreachable;
    try std.testing.expectEqual(@as(u32, 255), top);
    const c: u8 = 'z';
    _ = c;
    _ = banner;
    var x: ?i32 = null;
    x = undefined;
}
