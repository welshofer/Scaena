import type { Request, Response } from 'express';

export enum Status {
  Active = 'active',
  Archived = 'archived',
}

export interface User {
  readonly id: number;
  name: string;
  status: Status;
  tags?: string[];
}

type Handler<T> = (req: Request, res: Response) => Promise<T>;

const DEFAULT_PAGE_SIZE = 20;

@Injectable()
export class UserService {
  private users = new Map<number, User>();

  constructor(private readonly prefix: string = '/users') {}

  async find(id: number): Promise<User | undefined> {
    return this.users.get(id);
  }

  list(page = 0): User[] {
    const start = page * DEFAULT_PAGE_SIZE;
    return [...this.users.values()].slice(start, start + DEFAULT_PAGE_SIZE);
  }
}

export const handler: Handler<User[]> = async (req, res) => {
  const service = new UserService();
  const page = Number(req.query.page ?? 0) as number;
  const users = service.list(page).filter((u): u is User => u.status === Status.Active);
  res.json({ users, next: `${req.path}?page=${page + 1}` });
  return users satisfies User[];
};

function isKey<T extends object>(obj: T, key: keyof any): key is keyof T {
  return key in obj;
}
