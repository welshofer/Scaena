import SwiftUI

/// A counter that remembers its best.
@MainActor
final class Counter: ObservableObject {
    @Published private(set) var count = 0
    @Published var best: Int?

    func increment(by step: Int = 1) {
        count += step
        if count > (best ?? .min) {
            best = count
        }
    }
}

struct CounterView: View {
    @StateObject private var counter = Counter()
    let title: String

    var body: some View {
        VStack(spacing: 12) {
            Text("\(title): \(counter.count)")
                .font(.largeTitle)
            Button("Add one") { counter.increment() }
            if let best = counter.best {
                Text("Best so far: \(best)")
            }
        }
    }
}
