//! A small cache with expiry.

use std::collections::HashMap;
use std::time::{Duration, Instant};

#[derive(Debug, Clone, PartialEq)]
pub struct Entry<'a, T> {
    value: T,
    key: &'a str,
    expires: Instant,
}

/// What can go wrong.
#[derive(Debug)]
pub enum CacheError {
    Missing(String),
    Expired { since: Duration },
}

pub struct Cache<'a, T> {
    entries: HashMap<&'a str, Entry<'a, T>>,
    ttl: Duration,
}

impl<'a, T: Clone> Cache<'a, T> {
    pub const DEFAULT_TTL: Duration = Duration::from_secs(60);

    pub fn new() -> Self {
        Self { entries: HashMap::new(), ttl: Self::DEFAULT_TTL }
    }

    pub fn get(&self, key: &str) -> Result<T, CacheError> {
        let entry = self.entries.get(key).ok_or_else(|| CacheError::Missing(key.to_string()))?;
        if Instant::now() > entry.expires {
            return Err(CacheError::Expired { since: entry.expires.elapsed() });
        }
        Ok(entry.value.clone())
    }
}
