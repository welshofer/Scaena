#include <iostream>
#include <vector>

namespace geometry {

template <typename T>
class Polygon {
public:
    explicit Polygon(std::vector<T> points) : points_(std::move(points)) {}

    [[nodiscard]] T perimeter() const noexcept {
        T total{};
        for (std::size_t i = 0; i < points_.size(); ++i) {
            total += distance(points_[i], points_[(i + 1) % points_.size()]);
        }
        return total;
    }

    virtual ~Polygon() = default;

private:
    std::vector<T> points_;
};

}  // namespace geometry

constexpr auto kLimit = 1'000'000;
const char *kBanner = R"banner(
  Raw "strings" keep \n as written
)banner";

int main() {
    auto square = geometry::Polygon<double>({0.0, 1.0, 2.5});
    auto twice = [&](int x) { return x * 2; };
    if (square.perimeter() > 3.0 && twice(kLimit) != 0) {
        std::cout << "big: " << kLimit << '\n';
    }
    int *p = nullptr;
    return p == nullptr ? 0 : 1;
}
