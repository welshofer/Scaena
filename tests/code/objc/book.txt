#import <Foundation/Foundation.h>
#import "Library.h"

NS_ASSUME_NONNULL_BEGIN

@interface Book : NSObject
@property (nonatomic, copy) NSString *title;
@property (nonatomic, assign) NSInteger pages;
- (instancetype)initWithTitle:(NSString *)title pages:(NSInteger)pages;
@end

@implementation Book

- (instancetype)initWithTitle:(NSString *)title pages:(NSInteger)pages {
    self = [super init];
    if (self) {
        _title = [title copy];
        _pages = pages;
    }
    return self;
}

- (NSString *)description {
    return [NSString stringWithFormat:@"%@ (%ld pages)", self.title, (long)self.pages];
}

@end

int main(int argc, const char *argv[]) {
    @autoreleasepool {
        Book *book = [[Book alloc] initWithTitle:@"Dune" pages:412];
        NSArray<Book *> *shelf = @[book];
        BOOL ready = shelf.count > 0 ? YES : NO;
        NSLog(@"%@ %d", book, ready);
        id nothing = nil;
    }
    return 0;
}

NS_ASSUME_NONNULL_END
